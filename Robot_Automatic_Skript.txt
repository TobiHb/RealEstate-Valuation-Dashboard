Convert Square Feet To Square Meters
    [Arguments]    ${square_feet_text}
    ${cleaned_text}    Evaluate    "${square_feet_text}".replace(",", "")    # Remove commas to be able to convert it into ints
    ${square_feet}    Convert To Number    ${cleaned_text}
    ${square_meters}    Evaluate    ${square_feet} * ${CONVERSION_FACTOR}
    RETURN    ${square_meters}

Process Single Redfin Link

    Open Browser To Redfin    ${url}
    ${price_text}    Extract Property Info    //*[@id="content"]/div[8]/div[2]/div[1]/div[1]/section/div/div/div/div[1]/div[2]/div/div/div/div/div[1]/div
    ${price_cleaned}    Evaluate    "${price_text}".replace("$", "").replace(",", "")
    ${price}    Convert To Number    ${price_cleaned}
    Log To Console    Price: ${price}

    ${beds_text}    Extract Property Info    //*[@id="content"]/div[8]/div[2]/div[1]/div[1]/section/div/div/div/div[1]/div[2]/div/div/div/div/div[2]/div
    ${beds_float}    Convert To Number    ${beds_text}
    ${beds}    Evaluate    int(${beds_float})
    Log To Console    Beds: ${beds}

    ${baths_text}    Extract Property Info    //*[@id="content"]/div[8]/div[2]/div[1]/div[1]/section/div/div/div/div[1]/div[2]/div/div/div/div/div[3]/div
    ${baths_float}    Convert To Number    ${baths_text}
     ${baths}    Evaluate    int(${baths_float})
    Log To Console    Baths: ${baths}

    ${ZIP}    Extract Property Info    //*[@id="content"]/div[8]/div[2]/div[1]/div[1]/section/div/div/div/div[1]/div[2]/div/div/div/header/div/h1/div[2]
    ${clean_ZIP}    Evaluate    "${ZIP}".split()[-1]
    Log To Console    ZIP: ${clean_ZIP}

    ${square_feet}    Extract Property Info    //*[@id="content"]/div[8]/div[2]/div[1]/div[1]/section/div/div/div/div[1]/div[2]/div/div/div/div/div[4]/span
    ${square_meters_float}    Convert Square Feet To Square Meters    ${square_feet}
    ${square_meters}    Evaluate    int(${square_meters_float})
    Log To Console    Square meters: ${square_meters}

    RETURN    ${price}    ${beds}    ${baths}    ${clean_ZIP}    ${square_meters}
